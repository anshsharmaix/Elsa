  <script>
    "use strict";

    /* This stores the Alpha Vantage endpoint and application settings. */
    const API_URL =
      "https://www.alphavantage.co/query?function=TIME_SERIES_WEEKLY_ADJUSTED&symbol=SYMBOL&apikey=KEY";

    const DAILY_LIMIT = 25;

    let allWeeks = [];
    let filteredWeeks = [];
    let currentSymbol = "";
    let charts = {};

    /* This creates today's date in the user's local browser timezone. */
    function todayString() {
      const date = new Date();
      const year = date.getFullYear();
      const month = String(date.getMonth() + 1).padStart(2, "0");
      const day = String(date.getDate()).padStart(2, "0");

      return `${year}-${month}-${day}`;
    }

    /* This gets a local-storage item without crashing if storage is unavailable. */
    function getStorage(key) {
      try {
        return localStorage.getItem(key);
      } catch (error) {
        return null;
      }
    }

    /* This saves a local-storage item without crashing the application. */
    function setStorage(key, value) {
      try {
        localStorage.setItem(key, value);
        return true;
      } catch (error) {
        return false;
      }
    }

    /* This removes a local-storage item safely. */
    function removeStorage(key) {
      try {
        localStorage.removeItem(key);
      } catch (error) {
        /* Nothing needs to be done here. */
      }
    }

    /* This keeps the daily API counter on the current date only. */
    function getApiCounter() {
      const today = todayString();
      const savedDate = getStorage("stockDashboardApiDate");

      if (savedDate !== today) {
        setStorage("stockDashboardApiDate", today);
        setStorage("stockDashboardApiCalls", "0");
        return 0;
      }

      const value = parseInt(getStorage("stockDashboardApiCalls") || "0", 10);

      return Number.isFinite(value) ? value : 0;
    }

    /* This increases the number of API requests made today. */
    function incrementApiCounter() {
      const current = getApiCounter();
      const next = current + 1;

      setStorage("stockDashboardApiDate", todayString());
      setStorage("stockDashboardApiCalls", String(next));

      updateCounter();
    }

    /* This updates the API counter shown on the page. */
    function updateCounter() {
      const count = getApiCounter();

      document.getElementById("counter").textContent =
        `API calls used today: ${count} of ${DAILY_LIMIT}`;
    }

    /* This changes the status message and its visual style. */
    function setStatus(message, type = "") {
      const status = document.getElementById("status");

      status.textContent = message;
      status.className = "status";

      if (type) {
        status.classList.add(type);
      }
    }

    /* This loads a saved API key only when the user previously chose to remember it. */
    function loadSavedKey() {
      const savedKey = getStorage("stockDashboardApiKey");

      if (savedKey) {
        document.getElementById("apiKey").value = savedKey;
        document.getElementById("rememberKey").checked = true;
      }
    }

    /* This saves or removes the API key according to the checkbox. */
    function handleKeyStorage() {
      const key = document.getElementById("apiKey").value.trim();
      const remember = document.getElementById("rememberKey").checked;

      if (remember && key) {
        setStorage("stockDashboardApiKey", key);
      } else {
        removeStorage("stockDashboardApiKey");
      }
    }

    /* This converts Alpha Vantage text values into real JavaScript numbers. */
    function numberValue(value) {
      const number = Number(value);
      return Number.isFinite(number) ? number : 0;
    }

    /* This converts Alpha Vantage JSON into sorted weekly records. */
    function parseAlphaVantageData(json) {
      const series = json["Weekly Adjusted Time Series"];

      if (!series || typeof series !== "object") {
        return [];
      }

      return Object.entries(series)
        .map(([date, values]) => ({
          date,
          open: numberValue(values["1. open"]),
          high: numberValue(values["2. high"]),
          low: numberValue(values["3. low"]),
          close: numberValue(values["4. close"]),
          adjustedClose: numberValue(values["5. adjusted close"]),
          volume: numberValue(values["6. volume"]),
          dividend: numberValue(values["7. dividend amount"])
        }))
        .sort((a, b) => a.date.localeCompare(b.date));
    }

    /* This returns the local-storage key used for one stock's cached response. */
    function cacheKey(symbol) {
      return `stockDashboardCache_${symbol.toUpperCase()}`;
    }

    /* This reads the cached response for a symbol if it was saved today. */
    function getCachedData(symbol) {
      const raw = getStorage(cacheKey(symbol));

      if (!raw) {
        return null;
      }

      try {
        const cached = JSON.parse(raw);

        if (
          cached &&
          cached.date === todayString() &&
          cached.json
        ) {
          return cached.json;
        }
      } catch (error) {
        return null;
      }

      return null;
    }

    /* This stores the complete Alpha Vantage response for today's date. */
    function saveCachedData(symbol, json) {
      const cache = {
        date: todayString(),
        json: json
      };

      setStorage(cacheKey(symbol), JSON.stringify(cache));
    }

    /* This validates common Alpha Vantage error responses. */
    function getApiErrorMessage(json, symbol) {
      if (json && json.Note) {
        return "Daily API limit reached. You can use cached stocks or try again tomorrow.";
      }

      if (json && json.Information) {
        const information = String(json.Information).toLowerCase();

        if (
          information.includes("demo") ||
          information.includes("demo key") ||
          information.includes("apikey") ||
          information.includes("api key") ||
          information.includes("rate")
        ) {
          if (information.includes("demo") && symbol !== "IBM") {
            return "The demo key works only for IBM. Please enter your own Alpha Vantage key.";
          }

          return "Alpha Vantage returned an API access or rate-limit message.";
        }

        return String(json.Information);
      }

      if (json && json["Error Message"]) {
        return "Could not find that symbol";
      }

      return null;
    }

    /* This fetches fresh data directly from Alpha Vantage. */
    async function fetchFromApi(symbol, key) {
      const count = getApiCounter();

      if (count >= DAILY_LIMIT) {
        throw new Error(
          "Daily API limit reached. You can use cached stocks or try again tomorrow."
        );
      }

      const url = API_URL
        .replace("SYMBOL", encodeURIComponent(symbol))
        .replace("KEY", encodeURIComponent(key));

      let response;

      try {
        response = await fetch(url, {
          method: "GET",
          cache: "no-store"
        });
      } catch (error) {
        throw new Error(
          "No internet connection or Alpha Vantage could not be reached."
        );
      }

      incrementApiCounter();

      if (!response.ok) {
        throw new Error(
          "Alpha Vantage could not be reached right now. Please try again."
        );
      }

      let json;

      try {
        json = await response.json();
      } catch (error) {
        throw new Error("Alpha Vantage returned an unreadable response.");
      }

      const apiError = getApiErrorMessage(json, symbol);

      if (apiError) {
        throw new Error(apiError);
      }

      const parsed = parseAlphaVantageData(json);

      if (!parsed.length) {
        throw new Error("Could not find that symbol");
      }

      saveCachedData(symbol, json);

      return json;
    }

    /* This loads data either from today's cache or directly from the API. */
    async function loadStock(forceRefresh = false) {
      const key = document.getElementById("apiKey").value.trim();
      const symbol = document.getElementById("symbol").value.trim().toUpperCase();

      if (!key) {
        setStatus("Please type your Alpha Vantage API key.", "error");
        return;
      }

      if (!symbol) {
        setStatus("Please type a stock symbol.", "error");
        return;
      }

      handleKeyStorage();

      if (!forceRefresh) {
        const cachedJson = getCachedData(symbol);

        if (cachedJson) {
          const parsed = parseAlphaVantageData(cachedJson);

          if (parsed.length) {
            currentSymbol = symbol;
            allWeeks = parsed;

            setStatus(
              `${symbol}: data loaded from today's browser cache. No API call was made.`,
              "success"
            );

            initializeDateFilter();
            applyFilter();

            return;
          }
        }
      }

      try {
        setStatus(
          forceRefresh
            ? `Refreshing ${symbol} from Alpha Vantage...`
            : `Loading ${symbol} from Alpha Vantage...`
        );

        const json = await fetchFromApi(symbol, key);
        const parsed = parseAlphaVantageData(json);

        currentSymbol = symbol;
        allWeeks = parsed;

        setStatus(
          `${symbol}: fresh data loaded from the Alpha Vantage API and cached for today.`,
          "success"
        );

        initializeDateFilter();
        applyFilter();

      } catch (error) {
        setStatus(
          error.message || "Something went wrong while loading the stock.",
          "error"
        );
      }
    }

    /* This returns a date string a chosen number of months before another date. */
    function subtractMonths(date, months) {
      const result = new Date(date);
      result.setMonth(result.getMonth() - months);

      return result;
    }

    /* This formats a JavaScript date as YYYY-MM-DD. */
    function formatDate(date) {
      const year = date.getFullYear();
      const month = String(date.getMonth() + 1).padStart(2, "0");
      const day = String(date.getDate()).padStart(2, "0");

      return `${year}-${month}-${day}`;
    }

    /* This sets the default date filter to the latest available year. */
    function initializeDateFilter() {
      if (!allWeeks.length) {
        return;
      }

      const latest = new Date(`${allWeeks[allWeeks.length - 1].date}T00:00:00`);
      const oneYearAgo = subtractMonths(latest, 12);

      const availableStart = allWeeks[0].date;
      const selectedStart = formatDate(oneYearAgo);

      document.getElementById("fromDate").value =
        selectedStart < availableStart ? availableStart : selectedStart;

      document.getElementById("toDate").value = allWeeks[allWeeks.length - 1].date;

      setActiveQuickButton("1Y");
    }

    /* This applies a quick date range without making an API call. */
    function applyQuickRange(range) {
      if (!allWeeks.length) {
        return;
      }

      const latestString = allWeeks[allWeeks.length - 1].date;
      const latest = new Date(`${latestString}T00:00:00`);

      let startDate;

      if (range === "3M") {
        startDate = subtractMonths(latest, 3);
      } else if (range === "6M") {
        startDate = subtractMonths(latest, 6);
      } else if (range === "1Y") {
        startDate = subtractMonths(latest, 12);
      } else if (range === "3Y") {
        startDate = subtractMonths(latest, 36);
      } else if (range === "5Y") {
        startDate = subtractMonths(latest, 60);
      } else if (range === "YTD") {
        startDate = new Date(latest.getFullYear(), 0, 1);
      } else {
        startDate = new Date(`${allWeeks[0].date}T00:00:00`);
      }

      let from = formatDate(startDate);

      if (from < allWeeks[0].date) {
        from = allWeeks[0].date;
      }

      document.getElementById("fromDate").value = from;
      document.getElementById("toDate").value = latestString;

      setActiveQuickButton(range);
      applyFilter();
    }

    /* This filters the already-loaded data in the browser only. */
    function applyFilter() {
      if (!allWeeks.length) {
        return;
      }

      let from = document.getElementById("fromDate").value;
      let to = document.getElementById("toDate").value;

      if (!from) {
        from = allWeeks[0].date;
      }

      if (!to) {
        to = allWeeks[allWeeks.length - 1].date;
      }

      if (from > to) {
        [from, to] = [to, from];

        document.getElementById("fromDate").value = from;
        document.getElementById("toDate").value = to;
      }

      filteredWeeks = allWeeks.filter(
        week => week.date >= from && week.date <= to
      );

      updateKPIs();
      updateCharts();
      updateTable();

      setActiveQuickButton("");
    }

    /* This calculates the average of a number array. */
    function average(values) {
      if (!values.length) {
        return 0;
      }

      return values.reduce((sum, value) => sum + value, 0) / values.length;
    }

    /* This calculates standard deviation using the population formula. */
    function standardDeviation(values) {
      if (values.length === 0) {
        return 0;
      }

      const mean = average(values);

      const variance = average(
        values.map(value => Math.pow(value - mean, 2))
      );

      return Math.sqrt(variance);
    }

    /* This calculates the maximum drawdown from weekly adjusted closes. */
    function calculateMaxDrawdown(data) {
      if (!data.length) {
        return 0;
      }

      let peak = data[0].adjustedClose;
      let maxDrawdown = 0;

      data.forEach(week => {
        if (week.adjustedClose > peak) {
          peak = week.adjustedClose;
        }

        if (peak > 0) {
          const drawdown =
            ((week.adjustedClose - peak) / peak) * 100;

          if (drawdown < maxDrawdown) {
            maxDrawdown = drawdown;
          }
        }
      });

      return maxDrawdown;
    }

    /* This calculates week-over-week percentage returns. */
    function calculateReturns(data) {
      return data.map((week, index) => {
        if (index === 0 || data[index - 1].adjustedClose === 0) {
          return null;
        }

        return (
          ((week.adjustedClose - data[index - 1].adjustedClose) /
            data[index - 1].adjustedClose) *
          100
        );
      });
    }

    /* This creates a moving average using the last ten available weeks. */
    function calculateMovingAverage(data, period = 10) {
      return data.map((_, index) => {
        const start = Math.max(0, index - period + 1);
        const values = data
          .slice(start, index + 1)
          .map(week => week.adjustedClose);

        return average(values);
      });
    }

    /* This formats prices for display. */
    function formatPrice(value) {
      if (!Number.isFinite(value)) {
        return "—";
      }

      return value.toLocaleString("en-IN", {
        minimumFractionDigits: 2,
        maximumFractionDigits: 2
      });
    }

    /* This formats large volumes in a readable way. */
    function formatVolume(value) {
      if (!Number.isFinite(value)) {
        return "—";
      }

      return value.toLocaleString("en-IN", {
        maximumFractionDigits: 0
      });
    }

    /* This updates all KPI cards for the selected date range. */
    function updateKPIs() {
      if (!filteredWeeks.length) {
        document.querySelectorAll(".kpi-value").forEach(element => {
          element.textContent = "—";
        });

        return;
      }

      const first = filteredWeeks[0];
      const last = filteredWeeks[filteredWeeks.length - 1];

      const periodReturn =
        first.adjustedClose !== 0
          ? ((last.adjustedClose - first.adjustedClose) /
              first.adjustedClose) *
            100
          : 0;

      const highs = filteredWeeks.map(week => week.high);
      const lows = filteredWeeks.map(week => week.low);
      const volumes = filteredWeeks.map(week => week.volume);
      const dividends = filteredWeeks.map(week => week.dividend);

      const returns = calculateReturns(filteredWeeks).filter(
        value => value !== null
      );

      const volatility = standardDeviation(returns);

      const drawdown = calculateMaxDrawdown(filteredWeeks);

      document.getElementById("kpiClose").textContent =
        formatPrice(last.adjustedClose);

      const returnElement = document.getElementById("kpiReturn");
      returnElement.textContent =
        `${periodReturn >= 0 ? "+" : ""}${periodReturn.toFixed(2)}%`;
      returnElement.className =
        `kpi-value ${periodReturn >= 0 ? "positive" : "negative"}`;

      document.getElementById("kpiHigh").textContent =
        formatPrice(Math.max(...highs));

      document.getElementById("kpiLow").textContent =
        formatPrice(Math.min(...lows));

      document.getElementById("kpiVolume").textContent =
        formatVolume(average(volumes));

      document.getElementById("kpiDividend").textContent =
        dividends.reduce((sum, value) => sum + value, 0).toFixed(4);

      document.getElementById("kpiVolatility").textContent =
        `${volatility.toFixed(2)}%`;

      const drawdownElement = document.getElementById("kpiDrawdown");
      drawdownElement.textContent =
        `${drawdown.toFixed(2)}%`;
      drawdownElement.className =
        `kpi-value ${drawdown < 0 ? "negative" : ""}`;

      document.getElementById("kpiWeeks").textContent =
        filteredWeeks.length;
    }

    /* This safely escapes values before placing them into a CSV file. */
    function csvEscape(value) {
      const text = String(value ?? "");

      if (
        text.includes(",") ||
        text.includes('"') ||
        text.includes("\n")
      ) {
        return `"${text.replace(/"/g, '""')}"`;
      }

      return text;
    }

    /* This downloads all currently filtered weeks as a CSV file. */
    function downloadCSV() {
      if (!filteredWeeks.length) {
        setStatus("There is no filtered data to download.", "error");
        return;
      }

      const returns = calculateReturns(filteredWeeks);

      const rows = [
        [
          "Date",
          "Open",
          "High",
          "Low",
          "Close",
          "Adjusted Close",
          "Volume",
          "Dividend",
          "Weekly Return %"
        ]
      ];

      filteredWeeks.forEach((week, index) => {
        rows.push([
          week.date,
          week.open,
          week.high,
          week.low,
          week.close,
          week.adjustedClose,
          week.volume,
          week.dividend,
          returns[index] === null
            ? ""
            : returns[index]
        ]);
      });

      const csv = rows
        .map(row => row.map(csvEscape).join(","))
        .join("\n");

      const blob = new Blob(
        ["\uFEFF" + csv],
        { type: "text/csv;charset=utf-8;" }
      );

      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");

      link.href = url;
      link.download =
        `${currentSymbol || "stock"}_dashboard_data.csv`;

      document.body.appendChild(link);
      link.click();
      link.remove();

      URL.revokeObjectURL(url);
    }
